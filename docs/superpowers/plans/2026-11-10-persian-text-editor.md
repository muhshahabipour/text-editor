# Persian/English Text Editor Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A native macOS plain-text editor with native tabs, RTL/LTR auto direction, Persian-aware find, Persian spell check, line numbers and a stats status bar.

**Architecture:** Pure logic (stats, Persian text normalization, encoding detection) lives in a local Swift package `EditorCore`, tested with `swift test` (fast, no Xcode UI). The app target is a document-based AppKit app (`NSDocument` + `NSTextView`) generated by XcodeGen, depending on `EditorCore`.

**Tech Stack:** Swift 5.9+, AppKit, TextKit 2, XCTest, XcodeGen, macOS 13+.

Spec: `docs/superpowers/specs/2026-11-10-persian-text-editor-design.md`

```swift
import Foundation

public struct TextStats: Equatable {
    public var words: Int
    public var characters: Int
    public var lines: Int
    public init(words: Int, characters: Int, lines: Int) {
        self.words = words; self.characters = characters; self.lines = lines
    }
}

public enum StatsCalculator {
    public static func stats(for text: String) -> TextStats {
        var words = 0
        text.enumerateSubstrings(in: text.startIndex..., options: [.byWords, .substringNotRequired]) { _, _, _, _ in
            words += 1
        }
        let lines = text.reduce(1) { $1 == "\n" ? $0 + 1 : $0 }
        return TextStats(words: words, characters: text.count, lines: lines)
    }

    /// 1-based line and column (column counted in UTF-16 units, matching NSTextView selection).
    public static func position(in text: String, utf16Offset: Int) -> (line: Int, column: Int) {
        let ns = text as NSString
        let offset = max(0, min(utf16Offset, ns.length))
        var line = 1
        var lineStart = 0
        var i = 0
        while i < offset {
            if ns.character(at: i) == 0x0A { line += 1; lineStart = i + 1 }
            i += 1
        }
        return (line, offset - lineStart + 1)
    }
}
```

- [ ] **Step 3: Implement**

Normalization is a 1:1 scalar replacement, so UTF-16 offsets in the normalized text map directly onto the original.

```swift
import Foundation

public enum PersianText {
    private static let letterMap: [Character: Character] = [
        "ي": "ی", "ى": "ی", "ك": "ک",
    ]

    public static func normalizeLetters(_ s: String) -> String {
        String(s.map { letterMap[$0] ?? $0 })
    }

    public static func toPersianDigits(_ s: String) -> String {
        let persian = Array("۰۱۲۳۴۵۶۷۸۹")
        return String(s.map { c in
            if let d = c.wholeNumberValue, c.isASCII { return persian[d] }
            return c
        })
    }

    /// Ranges (UTF-16, in the ORIGINAL text) where `query` occurs.
    public static func findRanges(of query: String, in text: String, ignoreVariants: Bool) -> [NSRange] {
        guard !query.isEmpty else { return [] }
        let t = ignoreVariants ? normalizeLetters(text) : text
        let q = ignoreVariants ? normalizeLetters(query) : query
        let ns = t as NSString
        var results: [NSRange] = []
        var search = NSRange(location: 0, length: ns.length)
        while search.length > 0 {
            let r = ns.range(of: q, options: [.caseInsensitive], range: search)
            if r.location == NSNotFound { break }
            results.append(r)
            let next = r.location + max(r.length, 1)
            search = NSRange(location: next, length: ns.length - next)
        }
        return results
    }
}
```

```swift
import Foundation

public enum TextEncodingDetector {
    public static let windows1256 = String.Encoding(
        rawValue: CFStringConvertEncodingToNSStringEncoding(
            CFStringEncoding(CFStringEncodings.windowsArabic.rawValue)))

    public static func decode(_ data: Data) -> (text: String, encoding: String.Encoding)? {
        if data.count >= 2, (data[0] == 0xFF && data[1] == 0xFE) || (data[0] == 0xFE && data[1] == 0xFF) {
            if let s = String(data: data, encoding: .utf16) { return (s, .utf16) }
        }
        var body = data
        if data.starts(with: [0xEF, 0xBB, 0xBF]) { body = data.dropFirst(3) }
        if let s = String(data: body, encoding: .utf8) { return (s, .utf8) }
        if let s = String(data: data, encoding: windows1256) { return (s, windows1256) }
        return nil
    }

    public static func decode(_ data: Data, using encoding: String.Encoding) -> String? {
        String(data: data, encoding: encoding)
    }
}
```

```swift
import AppKit
import EditorCore

final class TextDocument: NSDocument {
    var text = ""
    private var encoding: String.Encoding = .utf8
    weak var editor: EditorViewController?

    override class var autosavesInPlace: Bool { true }

    override func makeWindowControllers() {
        addWindowController(EditorWindowController(document: self))
    }

    override func data(ofType typeName: String) throws -> Data {
        if let s = editor?.currentText { text = s }
        guard let d = text.data(using: encoding) else {
            throw NSError(domain: NSCocoaErrorDomain, code: NSFileWriteInapplicableStringEncodingError)
        }
        return d
    }

    override func read(from data: Data, ofType typeName: String) throws {
        if let r = TextEncodingDetector.decode(data) {
            text = r.text; encoding = r.encoding
        } else {
            throw NSError(domain: NSCocoaErrorDomain, code: NSFileReadInapplicableStringEncodingError,
                          userInfo: [NSLocalizedDescriptionKey:
                            "This file's text encoding could not be detected (tried UTF-8, UTF-16, Windows-1256)."])
        }
    }
}
```

```swift
import AppKit
import EditorCore

final class EditorViewController: NSViewController, NSTextViewDelegate {
    let document: TextDocument
    let scrollView = NSScrollView()
    let textView = NSTextView(usingTextLayoutManager: true)!

    var currentText: String { textView.string }

    init(document: TextDocument) {
        self.document = document
        super.init(nibName: nil, bundle: nil)
    }
    required init?(coder: NSCoder) { fatalError() }

    override func loadView() {
        let root = NSView()
        scrollView.hasVerticalScroller = true
        scrollView.translatesAutoresizingMaskIntoConstraints = false
        root.addSubview(scrollView)
        NSLayoutConstraint.activate([
            scrollView.topAnchor.constraint(equalTo: root.topAnchor),
            scrollView.leadingAnchor.constraint(equalTo: root.leadingAnchor),
            scrollView.trailingAnchor.constraint(equalTo: root.trailingAnchor),
            scrollView.bottomAnchor.constraint(equalTo: root.bottomAnchor),
        ])
        configureTextView()
        scrollView.documentView = textView
        view = root
    }

    private func configureTextView() {
        textView.isRichText = false
        textView.allowsUndo = true
        textView.isAutomaticQuoteSubstitutionEnabled = false
        textView.isAutomaticDashSubstitutionEnabled = false
        textView.isAutomaticTextReplacementEnabled = false
        textView.usesFindBar = true
        textView.isIncrementalSearchingEnabled = true
        textView.isContinuousSpellCheckingEnabled = true
        textView.textContainerInset = NSSize(width: 12, height: 10)
        textView.isVerticallyResizable = true
        textView.isHorizontallyResizable = false
        textView.autoresizingMask = [.width]
        textView.textContainer?.widthTracksTextView = true
        textView.baseWritingDirection = .natural
        textView.font = Self.defaultFont()
        textView.defaultParagraphStyle = {
            let p = NSMutableParagraphStyle()
            p.baseWritingDirection = .natural
            p.alignment = .natural
            return p
        }()
        textView.typingAttributes = [
            .font: textView.font!,
            .paragraphStyle: textView.defaultParagraphStyle!,
        ]
        textView.string = document.text
        textView.delegate = self
    }

    static func defaultFont() -> NSFont {
        NSFont(name: "Vazirmatn", size: 15) ?? NSFont.systemFont(ofSize: 15)
    }

    func textDidChange(_ notification: Notification) {
        document.updateChangeCount(.changeDone)
    }
}
```
`updateChangeCount(.changeDone)` is used because the text view's own undo manager is the window's undo manager only if `document.undoManager` is shared. Make that true by overriding in `EditorWindowController` is not needed: `NSTextView` uses `undoManager` from its window's delegate (the document), so edits register undo with the document and mark it dirty automatically. If the title bar shows the dirty dot twice or Undo doesn't clear it, remove the manual `updateChangeCount` call.

### Task 8: Line numbers

**Files:**
- Create: `App/Sources/LineNumberRulerView.swift`
- Modify: `App/Sources/EditorViewController.swift`, `App/Sources/MenuBuilder.swift`

- [ ] **Step 1: `LineNumberRulerView.swift`**

Uses TextKit 2 line fragments. Numbers are drawn per paragraph (logical line), on the left edge of the ruler (an `NSRulerView` is always on the left; spec said "leading side for RTL" — keeping it on the left for all text is the macOS-standard behavior and simpler; the spec is amended in Task 11).

```swift
import AppKit

final class LineNumberRulerView: NSRulerView {
    private var textView: NSTextView { clientView as! NSTextView }

    init(textView: NSTextView) {
        super.init(scrollView: textView.enclosingScrollView!, orientation: .verticalRuler)
        clientView = textView
        ruleThickness = 40
        NotificationCenter.default.addObserver(self, selector: #selector(setNeedsRedraw),
            name: NSText.didChangeNotification, object: textView)
        NotificationCenter.default.addObserver(self, selector: #selector(setNeedsRedraw),
            name: NSView.boundsDidChangeNotification, object: textView.enclosingScrollView!.contentView)
        textView.enclosingScrollView!.contentView.postsBoundsChangedNotifications = true
    }
    required init(coder: NSCoder) { fatalError() }

    @objc private func setNeedsRedraw() { needsDisplay = true }

    override func drawHashMarksAndLabels(in rect: NSRect) {
        NSColor.windowBackgroundColor.setFill(); bounds.fill()
        guard let tlm = textView.textLayoutManager,
              let tcm = tlm.textContentManager else { return }
        let attrs: [NSAttributedString.Key: Any] = [
            .font: NSFont.monospacedDigitSystemFont(ofSize: 11, weight: .regular),
            .foregroundColor: NSColor.secondaryLabelColor,
        ]
        let visible = textView.visibleRect
        let inset = textView.textContainerOrigin
        var line = 1
        tlm.enumerateTextLayoutFragments(from: tcm.documentRange.location,
                                         options: [.ensuresLayout]) { fragment in
            let frame = fragment.layoutFragmentFrame
            if frame.maxY >= visible.minY - 1, frame.minY <= visible.maxY {
                let y = frame.minY + inset.y - visible.minY
                let s = NSAttributedString(string: "\(line)", attributes: attrs)
                let size = s.size()
                s.draw(at: NSPoint(x: self.ruleThickness - size.width - 6, y: y))
            }
            line += 1
            return frame.minY <= visible.maxY
        }
    }
}
```
One fragment per paragraph in TextKit 2, which is exactly one number per logical line (wrapped lines are not numbered).

- [ ] **Step 1: `PersianFindController.swift`**

```swift
import AppKit
import EditorCore

final class PersianFindController: NSWindowController {
    private let findField = NSTextField()
    private let replaceField = NSTextField()
    private let ignoreCheck = NSButton(checkboxWithTitle: "Ignore ي/ی and ك/ک differences", target: nil, action: nil)
    private let status = NSTextField(labelWithString: "")
    weak var textView: NSTextView?

    convenience init() {
        let w = NSPanel(contentRect: NSRect(x: 0, y: 0, width: 420, height: 170),
                        styleMask: [.titled, .closable, .utilityWindow], backing: .buffered, defer: false)
        w.title = "Find (Persian-aware)"
        w.isFloatingPanel = true
        self.init(window: w)
        build()
    }

    private func build() {
        let content = window!.contentView!
        findField.placeholderString = "Find"; replaceField.placeholderString = "Replace with"
        ignoreCheck.state = Preferences.ignoreVariants ? .on : .off
        ignoreCheck.target = self; ignoreCheck.action = #selector(optionChanged)
        func btn(_ t: String, _ a: Selector) -> NSButton { NSButton(title: t, target: self, action: a) }
        let row = NSStackView(views: [btn("Previous", #selector(prev)), btn("Next", #selector(next)),
                                      btn("Replace", #selector(replaceOne)), btn("Replace All", #selector(replaceAll))])
        let stack = NSStackView(views: [findField, replaceField, ignoreCheck, row, status])
        stack.orientation = .vertical; stack.alignment = .leading; stack.spacing = 8
        stack.translatesAutoresizingMaskIntoConstraints = false
        content.addSubview(stack)
        NSLayoutConstraint.activate([
            stack.topAnchor.constraint(equalTo: content.topAnchor, constant: 12),
            stack.leadingAnchor.constraint(equalTo: content.leadingAnchor, constant: 12),
            stack.trailingAnchor.constraint(equalTo: content.trailingAnchor, constant: -12),
            findField.widthAnchor.constraint(equalTo: stack.widthAnchor),
            replaceField.widthAnchor.constraint(equalTo: stack.widthAnchor),
        ])
    }

    @objc private func optionChanged() { Preferences.ignoreVariants = ignoreCheck.state == .on }

    private func matches() -> [NSRange] {
        guard let tv = textView else { return [] }
        return PersianText.findRanges(of: findField.stringValue, in: tv.string, ignoreVariants: Preferences.ignoreVariants)
    }

    private func select(_ r: NSRange) {
        guard let tv = textView else { return }
        tv.setSelectedRange(r); tv.scrollRangeToVisible(r); tv.showFindIndicator(for: r)
    }

    @objc private func next() { step(forward: true) }
    @objc private func prev() { step(forward: false) }

    private func step(forward: Bool) {
        let all = matches()
        status.stringValue = "\(all.count) matches"
        guard let tv = textView, !all.isEmpty else { NSSound.beep(); return }
        let cur = tv.selectedRange()
        let target: NSRange
        if forward {
            target = all.first { $0.location >= cur.location + (cur.length > 0 ? 1 : 0) } ?? all[0]
        } else {
            target = all.last { $0.location < cur.location } ?? all[all.count - 1]
        }
        select(target)
    }

    @objc private func replaceOne() {
        guard let tv = textView else { return }
        let sel = tv.selectedRange()
        if matches().contains(sel), tv.shouldChangeText(in: sel, replacementString: replaceField.stringValue) {
            tv.replaceCharacters(in: sel, with: replaceField.stringValue)
            tv.didChangeText()
        }
        next()
    }

    @objc private func replaceAll() {
        guard let tv = textView else { return }
        let all = matches()
        guard !all.isEmpty else { NSSound.beep(); return }
        let rep = replaceField.stringValue
        tv.undoManager?.beginUndoGrouping()
        for r in all.reversed() where tv.shouldChangeText(in: r, replacementString: rep) {
            tv.replaceCharacters(in: r, with: rep)
        }
        tv.didChangeText()
        tv.undoManager?.endUndoGrouping()
        status.stringValue = "Replaced \(all.count)"
    }

    func show(for textView: NSTextView) {
        self.textView = textView
        showWindow(nil)
        window?.makeFirstResponder(findField)
    }
}
```
